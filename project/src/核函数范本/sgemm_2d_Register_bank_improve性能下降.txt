__global__ void __launch_bounds__(256, 2) sgemm_2d_Register_bank_improve(
    int M, int N, int K,
    const float* __restrict__ A,
    const float* __restrict__ B,
    float* __restrict__ C) 
{
    // -------------------------------------------------------------------------
    // 1. 基础分块参数与共享内存别名定义 (Union 内存复用)
    // -------------------------------------------------------------------------
    constexpr int BM = 128;
    constexpr int BN = 128;
    constexpr int BK = 16;
    constexpr int TM = 8;
    constexpr int TN = 8;

    // 主循环物理总占用: s_a(2x128x16x4B=16KB) + s_b(2x16x128x4B=16KB) = 32KB
    // 写回阶段物理总占用: s_out(2x32x128x4B=32KB)
    // 完美 1:1 别名复用，零追加内存开销！
    union SharedStorage {
        struct {
            float s_a[2][BM][BK];
            float s_b[2][BK][BN];
        } compute;
        float s_out[2][32][128]; // 双缓冲写回 Pipeline: 2 阶段 x 32 行 x 128 列
    };

    __shared__ SharedStorage smem;

    const int tid = threadIdx.x; // 0 ~ 255

    // 计算当前线程负责的 8x8 计算 Tile 在 Block 内的起始位置
    const int thread_row = (tid / 16) * TM; // 0, 8, 16 ... 120
    const int thread_col = (tid % 16) * TN; // 0, 8, 16 ... 120

    const int block_row = blockIdx.y * BM;
    const int block_col = blockIdx.x * BN;

    // 调整全局基指针到当前 Block 起始位置
    A += block_row * K;
    B += block_col;

    // 累加寄存器 (8x8)
    float accum[TM][TN] = {0.0f};


    // -------------------------------------------------------------------------
    // 2. Cooperative Load 索引映射 (按 float4 16字节向量计数)
    // -------------------------------------------------------------------------
    const int a_tile_row = tid / 4;        // 0 ~ 63
    const int a_tile_col4 = (tid % 4) * 4; // 0, 4, 8, 12
    const int b_tile_row = tid / 32;       // 0 ~ 7
    const int b_tile_col4 = (tid % 32) * 4;// 0, 4, 8 ... 124

    // -------------------------------------------------------------------------
    // 3. 流水线预热 (Prologue): 异步加载第 0 块数据到 Buffer 0
    // -------------------------------------------------------------------------
    int write_stage = 0;

    // 预加载 A (Buffer 0) + Swizzle
    #pragma unroll
    for (int i = 0; i < BM; i += 64) {
        int r = a_tile_row + i;
        int c = a_tile_col4;
        cp_async_cg(&smem.compute.s_a[write_stage][r][swizzle_a(r, c)], A + r * K + c);
    }

    // 预加载 B (Buffer 0) + Swizzle
    #pragma unroll
    for (int i = 0; i < BK; i += 8) {
        int r = b_tile_row + i;
        int c = b_tile_col4;
        cp_async_cg(&smem.compute.s_b[write_stage][r][swizzle_b(r, c)], B + r * N + c);
    }

    // 提交异步加载组并等待完成
    asm volatile("cp.async.commit_group;\n");
    asm volatile("cp.async.wait_group 0;\n");
    __syncthreads();

    // -------------------------------------------------------------------------
    // 4. Main Loop: 乒乓双缓冲计算与搬运重叠
    // -------------------------------------------------------------------------
    for (int k_step = BK; k_step < K; k_step += BK) {
        // 推进全局指针
        A += BK;
        B += BK * N;

        // 切换写入缓冲区索引 (0 -> 1, 1 -> 0)
        write_stage ^= 1;

        // --- Step 4A: 异步发射下一轮 (k_step) 数据到 write_stage 缓冲区 ---
        #pragma unroll
        for (int i = 0; i < BM; i += 64) {
            int r = a_tile_row + i;
            int c = a_tile_col4;
            cp_async_cg(&smem.compute.s_a[write_stage][r][swizzle_a(r, c)], A + r * K + c);
        }

        #pragma unroll
        for (int i = 0; i < BK; i += 8) {
            int r = b_tile_row + i;
            int c = b_tile_col4;
            cp_async_cg(&smem.compute.s_b[write_stage][r][swizzle_b(r, c)], B + r * N + c);
        }

        // 提交当前发射的异步加载组
        asm volatile("cp.async.commit_group;\n");

        // --- Step 4B: 计算当前 read_stage 缓冲区的数据 ---
        int read_stage = write_stage ^ 1;

        #pragma unroll
        for (int k = 0; k < BK; ++k) {
            #pragma unroll
            for (int i = 0; i < TM; ++i) {
                int r_a = thread_row + i;
                float a_val = smem.compute.s_a[read_stage][r_a][swizzle_a(r_a, k)];
                #pragma unroll
                for (int j = 0; j < TN; ++j) {
                    int c_b = thread_col + j;
                    accum[i][j] += a_val * smem.compute.s_b[read_stage][k][swizzle_b(k, c_b)];
                }
            }
        }

        // --- Step 4C: 等待下一次所需的数据传输完成 ---
        asm volatile("cp.async.wait_group 0;\n");
        __syncthreads();
    }

    // -------------------------------------------------------------------------
    // 5. Epilogue Loop: 处理最后一轮留在 Buffer 里的数据
    // -------------------------------------------------------------------------
    int read_stage = write_stage;
    #pragma unroll
    for (int k = 0; k < BK; ++k) {
        #pragma unroll
        for (int i = 0; i < TM; ++i) {
            int r_a = thread_row + i;
            float a_val = smem.compute.s_a[read_stage][r_a][swizzle_a(r_a, k)];
            #pragma unroll
            for (int j = 0; j < TN; ++j) {
                int c_b = thread_col + j;
                accum[i][j] += a_val * smem.compute.s_b[read_stage][k][swizzle_b(k, c_b)];
            }
        }
    }

    // -------------------------------------------------------------------------
    // 6. 极致写回 (Epilogue): SMEM 复用 + 双缓冲乒乓写回 + Swizzle
    // -------------------------------------------------------------------------
    __syncthreads(); // 彻底结束主计算阶段对 SMEM 的读取

    // --- 辅助 Lambda A: 将寄存器 fragment 写入指定 stage 的 SMEM (带 Swizzle) ---
    auto write_reg_to_smem = [&](int chunk, int stage) {
        int chunk_row_start = chunk * 32;
        #pragma unroll
        for (int i = 0; i < TM; ++i) {
            int local_r = thread_row + i; // 0 ~ 127
            if (local_r >= chunk_row_start && local_r < chunk_row_start + 32) {
                int smem_r = local_r - chunk_row_start; // 相对行号 0 ~ 31
                #pragma unroll
                for (int j = 0; j < TN; ++j) {
                    int smem_c = thread_col + j; // 相对列号 0 ~ 127
                    // swizzle_b 打散写入，避免 Bank 冲突
                    smem.s_out[stage][smem_r][swizzle_b(smem_r, smem_c)] = accum[i][j];
                }
            }
        }
    };

    // --- 辅助 Lambda B: 从指定 stage 的 SMEM 向量化读出并写回 GMEM (Coalesced float4) ---
    auto store_smem_to_gmem = [&](int chunk, int stage) {
        int chunk_row_start = chunk * 32;
        int write_r = tid / 8;                // 0 ~ 31 行
        int write_c_vec = (tid % 8) * 4;      // 0, 4, 8 ... 28 向量列 (对应 scalar 0, 16 ... 112)

        int global_c_row = block_row + chunk_row_start + write_r;
        int global_c_col = block_col + write_c_vec;

        // 从 SMEM 批量读出 (带 Swizzle，无 Bank 冲突)
        float4 val = *reinterpret_cast<float4*>(
            &smem.s_out[stage][write_r][swizzle_b(write_r, write_c_vec)]
        );

        // 打入全局内存 (100% Coalesced GMEM Store)
        if (global_c_row < M && global_c_col + 3 < N) {
            *reinterpret_cast<float4*>(&C[global_c_row * N + global_c_col]) = val;
        } else {
            // 边缘边界处理
            #pragma unroll
            for (int vec = 0; vec < 4; ++vec) {
                if (global_c_row < M && (global_c_col + vec) < N) {
                    float* val_arr = reinterpret_cast<float*>(&val);
                    C[global_c_row * N + global_c_col + vec] = val_arr[vec];
                }
            }
        }
    };

    // =========================================================================
    // 流水线阶段 1: Prologue - 预写 Chunk 0 到 Buffer 0
    // =========================================================================
    write_reg_to_smem(0, 0);
    __syncthreads(); // 屏障：确保 Chunk 0 完全就绪于 Buffer 0

    // =========================================================================
    // 流水线阶段 2: Main Pipeline Loop - 重叠处理 Chunk 0~2 显存写回 与 Chunk 1~3 SMEM 填充
    // =========================================================================
    #pragma unroll
    for (int chunk = 0; chunk < 3; ++chunk) {
        int cur_stage = chunk % 2;        // 当前可供写回 GMEM 的 Buffer (0 或 1)
        int next_stage = (chunk + 1) % 2; // 下一轮准备填入 SMEM 的 Buffer (1 或 0)
        int next_chunk = chunk + 1;

        // 1. 预写下一个 Chunk 到 next_stage Buffer (写入 SMEM)
        write_reg_to_smem(next_chunk, next_stage);

        // 2. 将当前 cur_stage Buffer 中的数据读出并写入 GMEM (并行/交错发射)
        store_smem_to_gmem(chunk, cur_stage);

        // 3. 屏障同步：
        //    a. 确保 next_stage 的 SMEM 写入彻底完成，可供下一轮安全读取
        //    b. 确保 cur_stage 的 SMEM 读取彻底完成，可供下下轮安全覆写
        __syncthreads();
    }

    // =========================================================================
    // 流水线阶段 3: Epilogue Drain - 收尾写回最后一个 Chunk 3
    // =========================================================================
    // Chunk 3 此时已经在 Buffer 1 (stage 1) 中就绪
    store_smem_to_gmem(3, 1);
}
